General Form of Functions function(arguments) { expression }

larger <- function(x, y) {
    if (any(x < 0)) 
        return(NA)
    y.is.bigger <- y > x
    x[y.is.bigger] <- y[y.is.bigger]
    x
}
larger(2, 5)
